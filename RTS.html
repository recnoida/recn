const url = "https://docs.google.com/spreadsheets/d/18ty35FnXzOq0_W0a4Z3MXP3Dy729Tfa2/gviz/tq?tqx=out:json&gid=310771952"; // Column headers to display const requiredHeaders = [ //"S.No", "Name", "SPORTS CIRCLE(2)", "MEN-METAL-MADNESS(3)", "EVENT (4)", "BREAKFAST RIDE (5)", "BROTHERHOOD NIGHT (7)", "EXPEDITION", "TOTAL" ]; fetch(url) .then(res => res.text()) .then(rep => { const json = JSON.parse(rep.substring(47).slice(0, -2)); const allCols = json.table.cols; const rows = json.table.rows; // Find column indexes by header names const selectedIndexes = requiredHeaders .map(header => allCols.findIndex(c => (c.label || "").trim() === header)) .filter(index => index !== -1); // Selected columns only const cols = selectedIndexes.map(i => allCols[i]); // Find Total column in original sheet let totalColIndex = allCols.findIndex(c => (c.label || "").toLowerCase().trim() === "total" ); let tableRows = []; rows.forEach(r => { let rowArr = []; let totalValue = 0; if (totalColIndex !== -1) { let tCell = r.c[totalColIndex]; totalValue = parseFloat(tCell ? (tCell.v || tCell.f) : 0) || 0; } // Read only selected columns selectedIndexes.forEach(i => { let cell = r.c[i]; let value = cell ? (cell.f || cell.v || "") : ""; rowArr.push(value); }); tableRows.push({ data: rowArr, total: totalValue }); }); // Sort high to low tableRows.sort((a, b) => b.total - a.total); // Header let headHTML = ""; cols.forEach(col => { headHTML += `${col.label || ""}`; }); headHTML += ""; document.getElementById("tableHead").innerHTML = headHTML; // Rows let bodyHTML = ""; tableRows.forEach(row => { let rowClass = ""; if (row.total === 0) rowClass = "red"; else if (row.total >= 40) rowClass = "green"; else if (row.total >= 30) rowClass = "yellow"; bodyHTML += ``; row.data.forEach(val => { bodyHTML += `${val}`; }); bodyHTML += ""; }); document.getElementById("sheetData").innerHTML = bodyHTML; }); // Search filter document.getElementById("searchInput").addEventListener("keyup", function() { let filter = this.value.toLowerCase(); document.querySelectorAll("#sheetData tr").forEach(row => { row.style.display = row.innerText.toLowerCase().includes(filter) ? "" : "none"; }); });
